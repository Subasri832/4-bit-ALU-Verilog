module alu4(
    input [3:0] A,
    input [3:0] B,
    input [2:0] opcode,
    output reg [3:0] Result,
    output reg Carry,
    output reg Zero,
    output reg Overflow
);

always @(*) begin

    // Default values
    Result = 4'b0000;
    Carry = 1'b0;
    Overflow = 1'b0;

    case(opcode)

        // Addition
        3'b000: begin
            {Carry, Result} = A + B;
            Overflow = (A[3] == B[3]) &&
                       (Result[3] != A[3]);
        end

        // Subtraction
        3'b001: begin
            Result = A - B;
        end

        // AND
        3'b010: begin
            Result = A & B;
        end

        // OR
        3'b011: begin
            Result = A | B;
        end

        // XOR
        3'b100: begin
            Result = A ^ B;
        end

        // NOT
        3'b101: begin
            Result = ~A;
        end

        // Left Shift
        3'b110: begin
            Result = A << 1;
        end

        // Right Shift
        3'b111: begin
            Result = A >> 1;
        end

    endcase

    // Zero flag
    if (Result == 4'b0000)
        Zero = 1'b1;
    else
        Zero = 1'b0;

end

endmodule
